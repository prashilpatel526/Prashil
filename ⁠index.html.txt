    <script>
        lucide.createIcons();

        // Real IST Clock
        function updateISTClock() {
            const now = new Date();
            const timeString = new Intl.DateTimeFormat('en-GB', { timeZone: 'Asia/Kolkata', hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' }).format(now);
            document.getElementById('ist-clock').innerText = timeString + ' IST';
        }
        setInterval(updateISTClock, 1000);
        updateISTClock();

        let currentSymbolCode = 'BTCUSDT';
        let currentTVsymbol = 'BINANCE:BTCUSDT';
        let currentTimeframe = '15';
        let tvWidget = null;
        let currentLivePrice = 64465.54;

        function initTVWidget(symbol, tvSym, tf) {
            currentSymbolCode = symbol;
            currentTVsymbol = tvSym;
            currentTimeframe = tf;
            document.getElementById('tv-pair-title').innerText = tvSym;
            document.getElementById('active-tf-badge').innerText = `Timeframe: ${tf} | Active Sync`;
            document.getElementById('tradingview-container').innerHTML = '';

            tvWidget = new TradingView.widget({
                "autosize": true,
                "symbol": tvSym,
                "interval": tf,
                "timezone": "Asia/Kolkata",
                "theme": "dark",
                "style": "1",
                "locale": "en",
                "toolbar_bg": "#04060a",
                "enable_publishing": false,
                "hide_side_toolbar": false,
                "allow_symbol_change": false,
                "container_id": "tradingview-container"
            });

            fetchLivePriceAndEngine();
        }

        async function fetchLivePriceAndEngine() {
            try {
                if (currentSymbolCode === 'XAUUSD') {
                    currentLivePrice = 2436.72;
                } else {
                    const res = await fetch(`https://api.binance.com/api/v3/ticker/price?symbol=${currentSymbolCode}`);
                    const data = await res.json();
                    currentLivePrice = parseFloat(data.price);
                }
                document.getElementById('header-price').innerText = '$' + currentLivePrice.toLocaleString('en-US', { minimumFractionDigits: 2 });
                updateEngineForTimeframe();
            } catch (err) {
                console.error("Price fetch error:", err);
                if (currentSymbolCode === 'XAUUSD') currentLivePrice = 2436.72;
                updateEngineForTimeframe();
            }
        }

        function updateEngineForTimeframe() {
            const isGold = currentSymbolCode === 'XAUUSD';
            if (isGold && (currentLivePrice > 5000 || currentLivePrice < 1000)) currentLivePrice = 2436.72;

            document.getElementById('matrix-title-label').innerText = `ICT & GLAZE MATRIX (${currentSymbolCode} - ${currentTimeframe} Timeframe)`;
            document.getElementById('backtest-table-heading').innerText = `HISTORICAL & LIVE CANDLE BACKTEST LOG (${currentSymbolCode} - ${currentTimeframe})`;
            document.getElementById('plan-header-title').innerText = `${currentSymbolCode} ${currentTimeframe} TRADE PLAN`;

            const biasBadge = document.getElementById('tf-bias-status-badge');
            const planBadge = document.getElementById('plan-bias-badge');

            biasBadge.innerText = 'Bias: BEARISH';
            biasBadge.className = 'text-red-400 bg-red-500/10 px-2 py-0.5 rounded text-[11px]';
            planBadge.innerText = 'LIVE SHORT';
            planBadge.className = 'text-red-400 bg-red-500/10 px-2 py-0.5 rounded text-[10px]';

            let dist = currentLivePrice * (isGold ? 0.003 : 0.006);
            let entry = currentLivePrice;
            let sl = entry + (dist * 0.5);
            let tp = entry - (dist * 0.9);

            document.getElementById('plan-entry').innerText = '$' + entry.toFixed(2);
            document.getElementById('plan-sl').innerText = '$' + sl.toFixed(2);
            document.getElementById('plan-tp').innerText = '$' + tp.toFixed(2);
